function bossShoot(){ if(!game.boss) return; const b = game.boss; const speed = getSpeedMultiplier(); const centerX = b.x + b.w / 2; const centerY = b.y + b.h / 2; let baseDelay = 400; switch(b.attackPattern % 3){ case 0: // Spiral Attack (Velocity reduced from 0.11 to 0.0825) const spiralBullets = 6; for(let i = 0; i < spiralBullets; i++){ const angle = b.spiralAngle + (i * (Math.PI * 2 / spiralBullets)); game.bullets.push({ x: centerX, y: centerY, vx: Math.cos(angle) * (0.0825 * speed), vy: Math.sin(angle) * (0.0825 * speed), owner: "boss" }); } b.spiralAngle += 0.28; baseDelay = 220 / speed; break; case 1: // 2-Bullet Spray Attack (Velocity reduced from 0.15 to 0.1125) const px = game.playerX + 8; const py = game.playerY + 6; const angleToPlayer = Math.atan2(py - centerY, px - centerX); const spread = [-0.15, 0.15]; for(let offset of spread){ const angle = angleToPlayer + offset; game.bullets.push({ x: centerX, y: centerY, vx: Math.cos(angle) * (0.1125 * speed), vy: Math.sin(angle) * (0.1125 * speed), owner: "boss" }); } baseDelay = 650 / speed; break; case 2: // Ring Attack (Velocity reduced from 0.09 to 0.0675) const ringBullets = 12; for(let i = 0; i < ringBullets; i++){ const angle = i * (Math.PI * 2 / ringBullets); game.bullets.push({ x: centerX, y: centerY, vx: Math.cos(angle) * (0.0675 * speed), vy: Math.sin(angle) * (0.0675 * speed), owner: "boss" }); } baseDelay = 950 / speed; break; } b.patternTimer++; if(b.patternTimer > 8){ b.patternTimer = 0; b.attackPattern++; } game.nextBossShootInterval = baseDelay + (Math.random() * 200); }
